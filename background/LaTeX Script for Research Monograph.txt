\section{Autogenerative Functors and Language Sheaves}
\begin{definition}[Autogenerative Functor]
Let $\mathsf{Seq}$ be the category of finite sequences with prefix inclusions. An endofunctor $\mathcal{A}:\mathsf{Seq}\to\mathsf{Prob}$ maps each sequence to a continuation law and each inclusion to a marginalization morphism. $\mathcal{A}$ is autogenerative if naturality squares are pullbacks.
\end{definition}
\begin{proposition}[Natural Sheaf Equivalence]
Every autogenerative functor defines a language sheaf; conversely, every flabby language sheaf induces an autogenerative functor up to natural isomorphism.
\end{proposition}
\section{Geodesic Coherence}
\begin{theorem}[Geodesic Coherence]
If generation minimizes expected prediction error under the autogenerative Fisher metric, then expected perplexity is minimized iff $\nabla_{\dot\gamma}\dot\gamma=0$.
\end{theorem}
\section{Cohomological Memory}
\begin{definition}[Memory Cohomology]
For sheaf $\mathcal{F}$ of conditionals, $H^k(\mathsf{Seq},\mathcal{F})$ with Čech coboundary encodes $k$-th order incompatibilities.
\end{definition}
\begin{proposition}
$H^0$ gives globally consistent distributions; $H^1$ measures local-to-global failure; $H^2$ captures higher-order ambiguity.
